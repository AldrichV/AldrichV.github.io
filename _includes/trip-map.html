{%- comment -%} Hand-drawn route map for a trip. Entries with `pin: [lat, lon]` become pins, joined in date order. Optional `pin_label: left` or `above` moves an entry's label (default: right of the pin). Usage: {% include trip-map.html trip="japan-2026" %} {%- endcomment -%} {%- assign _entries = site.journal | where: 'trip', include.trip | sort: 'date' -%} {%- assign _d = "" -%} {%- for e in _entries -%}{%- if e.pin -%} {%- assign _x = e.pin[1] | minus: 129.4 | times: 48.541 | plus: 8 | round: 1 -%} {%- assign _y = 41.7 | minus: e.pin[0] | times: 60 | plus: 8 | round: 1 -%} {%- if _d == "" -%}{%- assign _d = "M" | append: _x | append: " " | append: _y -%}{%- else -%}{%- assign _d = _d | append: " L" | append: _x | append: " " | append: _y -%}{%- endif -%} {%- endif -%}{%- endfor -%} {%- if _d != "" -%} {%- endif -%}
